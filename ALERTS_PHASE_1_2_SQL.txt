BEGIN;

ALTER TABLE public.alerts
ADD COLUMN IF NOT EXISTS category character varying(30) NOT NULL DEFAULT 'SYSTEM',
ADD COLUMN IF NOT EXISTS context_key character varying(180),
ADD COLUMN IF NOT EXISTS requires_action boolean NOT NULL DEFAULT false,
ADD COLUMN IF NOT EXISTS due_at timestamp with time zone,
ADD COLUMN IF NOT EXISTS read_at timestamp with time zone,
ADD COLUMN IF NOT EXISTS snoozed_until timestamp with time zone,
ADD COLUMN IF NOT EXISTS expires_at timestamp with time zone,
ADD COLUMN IF NOT EXISTS priority smallint NOT NULL DEFAULT 50;

ALTER TABLE public.alerts
DROP CONSTRAINT IF EXISTS alerts_type_check;

ALTER TABLE public.alerts
ADD CONSTRAINT alerts_type_check
CHECK (type IN (
    'SHORTAGE',
    'CRITICAL_STOCK',
    'SURPLUS_AVAILABLE',
    'BUY_OPPORTUNITY',
    'SELL_SUGGESTION',
    'EXTERNAL_DEFICIT',
    'NEED_UNDERCOVERED',
    'NEED_RESPONSE_RECEIVED',
    'NEED_DEADLINE_APPROACHING',
    'OFFER_REJECTED',
    'ORDER_REQUIRES_CONFIRMATION',
    'ORDER_DELIVERY_OVERDUE',
    'ORDER_PURCHASE_CREATED',
    'ORDER_CONFIRMED',
    'ORDER_IN_PROGRESS',
    'ORDER_DELIVERING',
    'ORDER_CANCELLED',
    'ORDER_COMPLETED',
    'LISTING_EXPIRING_SOON',
    'MESSAGE_UNREAD'
));

ALTER TABLE public.alerts
DROP CONSTRAINT IF EXISTS alerts_category_check;

ALTER TABLE public.alerts
ADD CONSTRAINT alerts_category_check
CHECK (category IN (
    'STOCK',
    'NEEDS',
    'ORDERS',
    'MARKETPLACE',
    'MESSAGES',
    'SYSTEM'
));

CREATE INDEX IF NOT EXISTS idx_alerts_producer_status_priority
ON public.alerts (producer_id, status, priority, updated_at DESC);

CREATE INDEX IF NOT EXISTS idx_alerts_producer_category_status
ON public.alerts (producer_id, category, status, updated_at DESC);

CREATE UNIQUE INDEX IF NOT EXISTS uniq_active_alert_context
ON public.alerts (producer_id, type, context_key)
WHERE status IN ('ACTIVE', 'READ', 'IGNORED', 'RESOLVED')
  AND cleared_at IS NULL
  AND context_key IS NOT NULL;

CREATE TABLE IF NOT EXISTS public.alert_deliveries
(
    id uuid NOT NULL DEFAULT gen_random_uuid(),
    alert_id uuid NOT NULL,
    user_id uuid NOT NULL,
    channel character varying(20) NOT NULL,
    status character varying(20) NOT NULL DEFAULT 'PENDING',
    error text,
    sent_at timestamp with time zone,
    created_at timestamp with time zone NOT NULL DEFAULT now(),
    CONSTRAINT alert_deliveries_pkey PRIMARY KEY (id),
    CONSTRAINT alert_deliveries_alert_id_fkey FOREIGN KEY (alert_id)
        REFERENCES public.alerts (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE,
    CONSTRAINT alert_deliveries_user_id_fkey FOREIGN KEY (user_id)
        REFERENCES public.users (id) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE,
    CONSTRAINT alert_deliveries_channel_check CHECK (channel IN ('IN_APP', 'EMAIL', 'SMS')),
    CONSTRAINT alert_deliveries_status_check CHECK (status IN ('PENDING', 'SENT', 'SKIPPED', 'FAILED'))
);

CREATE INDEX IF NOT EXISTS idx_alert_deliveries_alert_channel
ON public.alert_deliveries (alert_id, channel);

CREATE INDEX IF NOT EXISTS idx_alert_deliveries_user_status
ON public.alert_deliveries (user_id, status, created_at DESC);

COMMIT;
