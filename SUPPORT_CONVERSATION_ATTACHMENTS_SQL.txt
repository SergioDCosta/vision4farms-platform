-- Executar primeiro na BD local e só depois no Railway.

ALTER TABLE support_tickets
ADD COLUMN IF NOT EXISTS last_message_at TIMESTAMPTZ NULL;

ALTER TABLE support_tickets
ADD COLUMN IF NOT EXISTS last_message_by_role VARCHAR(20) NULL;

DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1
    FROM pg_constraint
    WHERE conname = 'support_tickets_last_message_by_role_check'
  ) THEN
    ALTER TABLE support_tickets
    ADD CONSTRAINT support_tickets_last_message_by_role_check
    CHECK (
      last_message_by_role IS NULL
      OR last_message_by_role IN ('REQUESTER', 'ADMIN', 'SYSTEM')
    );
  END IF;
END $$;

CREATE TABLE IF NOT EXISTS support_ticket_messages (
  id UUID PRIMARY KEY,
  ticket_id UUID NOT NULL REFERENCES support_tickets(id) ON DELETE CASCADE,
  sender_user_id UUID NULL REFERENCES users(id) ON DELETE SET NULL,
  sender_role VARCHAR(20) NOT NULL CHECK (sender_role IN ('REQUESTER', 'ADMIN', 'SYSTEM')),
  body TEXT NOT NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS support_ticket_attachments (
  id UUID PRIMARY KEY,
  message_id UUID NOT NULL REFERENCES support_ticket_messages(id) ON DELETE CASCADE,
  storage_path TEXT NOT NULL,
  file_name VARCHAR(255) NOT NULL,
  content_type VARCHAR(100) NOT NULL,
  size_bytes BIGINT NOT NULL DEFAULT 0,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_support_ticket_messages_ticket_created
ON support_ticket_messages (ticket_id, created_at);

CREATE INDEX IF NOT EXISTS idx_support_ticket_messages_sender_role
ON support_ticket_messages (ticket_id, sender_role, created_at);

CREATE INDEX IF NOT EXISTS idx_support_ticket_attachments_message
ON support_ticket_attachments (message_id);

CREATE INDEX IF NOT EXISTS idx_support_tickets_last_message_role
ON support_tickets (status, last_message_by_role, last_message_at);

-- Backfill leve para tickets antigos. A thread nova continua vazia nesses tickets,
-- mas a UI usa fallback com support_tickets.message/admin_reply_message.
UPDATE support_tickets
SET
  last_message_at = COALESCE(last_message_at, admin_replied_at, closed_at, updated_at, created_at),
  last_message_by_role = COALESCE(
    last_message_by_role,
    CASE
      WHEN admin_reply_message IS NOT NULL AND admin_reply_message <> '' THEN 'ADMIN'
      ELSE 'REQUESTER'
    END
  )
WHERE last_message_at IS NULL
   OR last_message_by_role IS NULL;
